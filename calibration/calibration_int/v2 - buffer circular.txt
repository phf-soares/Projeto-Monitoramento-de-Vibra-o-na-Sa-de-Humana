#include <stdio.h>
#include "driver/i2c.h"
#include "esp_log.h"
#include "esp_timer.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/ringbuf.h"

#define TAG "BNO055_STREAM"

#define BNO055_ADDR                0x28
#define BNO055_ACCEL_DATA_X_LSB    0x08

#define I2C_MASTER_SCL_IO          22
#define I2C_MASTER_SDA_IO          21
#define I2C_MASTER_PORT            I2C_NUM_0
#define I2C_MASTER_FREQ_HZ         400000

typedef struct {
    uint32_t timestamp_us;
    float ax, ay, az;
} accel_sample_t;

static RingbufHandle_t accel_ringbuf;
static esp_timer_handle_t sample_timer;

// ==================================================================
// I2C FUNCTIONS
// ==================================================================
static esp_err_t i2c_master_init(void)
{
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER,
        .sda_io_num = I2C_MASTER_SDA_IO,
        .scl_io_num = I2C_MASTER_SCL_IO,
        .sda_pullup_en = GPIO_PULLUP_ENABLE,
        .scl_pullup_en = GPIO_PULLUP_ENABLE,
        .master.clk_speed = I2C_MASTER_FREQ_HZ,
    };
    ESP_ERROR_CHECK(i2c_param_config(I2C_MASTER_PORT, &conf));
    return i2c_driver_install(I2C_MASTER_PORT, conf.mode, 0, 0, 0);
}

static esp_err_t bno055_read_len(uint8_t reg, uint8_t *data, uint8_t len)
{
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (BNO055_ADDR << 1) | I2C_MASTER_WRITE, true);
    i2c_master_write_byte(cmd, reg, true);
    i2c_master_start(cmd);
    i2c_master_write_byte(cmd, (BNO055_ADDR << 1) | I2C_MASTER_READ, true);
    if (len > 1)
        i2c_master_read(cmd, data, len - 1, I2C_MASTER_ACK);
    i2c_master_read_byte(cmd, data + len - 1, I2C_MASTER_NACK);
    i2c_master_stop(cmd);
    esp_err_t ret = i2c_master_cmd_begin(I2C_MASTER_PORT, cmd, 1000 / portTICK_PERIOD_MS);
    i2c_cmd_link_delete(cmd);
    return ret;
}

// ==================================================================
// SENSOR READ (FAST)
// ==================================================================
static void IRAM_ATTR sample_accel_callback(void *arg)
{
    uint8_t buf[6];
    if (bno055_read_len(BNO055_ACCEL_DATA_X_LSB, buf, 6) != ESP_OK) return;

    int16_t x = ((int16_t)buf[1] << 8) | buf[0];
    int16_t y = ((int16_t)buf[3] << 8) | buf[2];
    int16_t z = ((int16_t)buf[5] << 8) | buf[4];

    accel_sample_t sample;
    sample.timestamp_us = esp_timer_get_time();
    sample.ax = x / 100.0f;
    sample.ay = y / 100.0f;
    sample.az = z / 100.0f;

    // envia para ring buffer (não bloqueia)
    BaseType_t ok = xRingbufferSendFromISR(accel_ringbuf, &sample, sizeof(sample), NULL);
    if (ok == pdFALSE) {
        // buffer cheio, pode descartar ou sinalizar
    }
}

// ==================================================================
// TASK DE LEITURA (CONSUMIDOR)
// ==================================================================
static void accel_consumer_task(void *arg)
{
    while (1) {
        size_t item_size;
        accel_sample_t *item = (accel_sample_t *)xRingbufferReceive(accel_ringbuf, &item_size, pdMS_TO_TICKS(1000));
        if (item != NULL) {
            ESP_LOGI(TAG, "[%lu us] X=%.2f Y=%.2f Z=%.2f", item->timestamp_us, item->ax, item->ay, item->az);
            vRingbufferReturnItem(accel_ringbuf, (void *)item);
        }
    }
}

// ==================================================================
// MAIN
// ==================================================================
void app_main(void)
{
    ESP_ERROR_CHECK(i2c_master_init());

    // Cria ring buffer para armazenar ~2 segundos (1000Hz * 2s * 16 bytes = 32kB)
    accel_ringbuf = xRingbufferCreate(32 * 1024, RINGBUF_TYPE_BYTEBUF);
    if (!accel_ringbuf) {
        ESP_LOGE(TAG, "Erro ao criar ring buffer");
        return;
    }

    // Cria tarefa para processar dados
    xTaskCreate(accel_consumer_task, "accel_consumer", 4096, NULL, 5, NULL);

    // Cria timer de amostragem (1 kHz = 1000 µs)
    const esp_timer_create_args_t timer_args = {
        .callback = &sample_accel_callback,
        .name = "accel_timer"
    };
    ESP_ERROR_CHECK(esp_timer_create(&timer_args, &sample_timer));
    ESP_ERROR_CHECK(esp_timer_start_periodic(sample_timer, 1000)); // 1000 µs = 1 kHz

    ESP_LOGI(TAG, "Captura contínua iniciada a 1000 Hz!");
}
